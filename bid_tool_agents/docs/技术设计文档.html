1. 项目概述

1.1 项目背景

随着我国公共资源交易市场化改革不断深化，招投标领域面临着日益严峻的合规监管挑战，包括串标围标、虚假资质、关联公司围标等问题。传统人工审查方式效率低、覆盖面窄、标准不统一。

1.2 建设目标

通过AI Agent技术实现招投标文件的智能审查，提高工作效率，减少人工疏漏，为决策提供全面参考意见。

1.3 建设原则

  • API-first：核心服务通过API暴露，支持多种客户端和外部系统集成
  • Agent架构：基于LangGraph实现多Agent协作，支持功能灵活扩展
  • 四层记忆机制：每个Agent具有独立的四层记忆能力
  • 安全可控：确保系统和数据安全，支持私有化部署
  • 循序渐进：分阶段建设，逐步完善
  • 人机协同：系统输出仅作为辅助参考，不替代法定责任主体

2.2 架构特性

  1. 前后端分离：Vue3前端通过REST API与后端通信
  2. 多Agent协作：基于LangGraph实现复杂的Agent工作流
  3. 四层记忆机制：每个Agent具有独立的记忆能力
  4. LLM可扩展：支持前期阿里云API、后期私有化部署
  5. 统一文档解析：基于Unstructured.io的文档处理
  6. 多数据库协同：PostgreSQL + Qdrant + Neo4j + Redis

5.2 Agent基类

from abc import ABC, abstractmethod
from typing import Any, Dict, Optional
from langgraph.graph import StateGraph
from .memory import AgentMemory

class BaseAgent(ABC):
    """Agent抽象基类，所有Agent继承此类"""

    def __init__(self, agent_id: str, name: str):
        self.agent_id = agent_id
        self.name = name
        self.memory = AgentMemory(agent_id)
        self.graph = self._build_graph()

    @abstractmethod
    def _build_graph(self) -> StateGraph:
        """构建LangGraph"""
        pass

    @abstractmethod
    async def process(self, input_data: Any) -> Dict:
        """处理输入"""
        pass

    def get_memory(self) -> AgentMemory:
        return self.memory

5.3 LangGraph示例（Coordinator）

from langgraph.graph import StateGraph, END
from langgraph.checkpoint.postgres import PostgresSaver
from .nodes import (
    intent_recognition_node,
    task_decomposition_node,
    agent_dispatch_node,
    result_aggregation_node
)
from .state import CoordinatorState

def create_coordinator_graph(checkpoint_db_url: str):
    """创建协调器Agent的LangGraph"""

    checkpointer = PostgresSaver.from_conn_string(checkpoint_db_url)
    workflow = StateGraph(CoordinatorState)

    # 添加节点
    workflow.add_node("intent_recognition", intent_recognition_node)
    workflow.add_node("task_decomposition", task_decomposition_node)
    workflow.add_node("agent_dispatch", agent_dispatch_node)
    workflow.add_node("result_aggregation", result_aggregation_node)

    # 添加边
    workflow.add_edge("intent_recognition", "task_decomposition")
    workflow.add_edge("task_decomposition", "agent_dispatch")
    workflow.add_edge("agent_dispatch", "result_aggregation")
    workflow.add_edge("result_aggregation", END)

    # 编译
    return workflow.compile(
        checkpointer=checkpointer,
        interrupt_before=["agent_dispatch"]  # 支持人工确认
    )

7. LLM可扩展架构

7.1 LLM类型定义

from enum import Enum
from dataclasses import dataclass
from typing import Optional

class LLMProvider(Enum):
    """LLM提供者类型"""
    ALI_QWEN = "ali_qwen"           # 阿里云API
    PRIVATE_QWEN = "private_qwen"   # 千问私有化
    PRIVATE_DEEPSEEK = "private_deepseek"

class LLMModel(Enum):
    """具体模型枚举"""
    QWEN_MAX = "qwen-max"
    QWEN_PLUS = "qwen-plus"
    QWEN_TURBO = "qwen-turbo"
    QWEN3_14B = "qwen3-14b"
    QWEN3_72B = "qwen3-72b"
    DEEPSEEK_V3 = "deepseek-v3"

@dataclass
class LLMConfig:
    """LLM配置"""
    provider: LLMProvider
    model: LLMModel
    api_base: Optional[str] = None
    api_key: Optional[str] = None
    max_tokens: int = 4096
    temperature: float = 0.7
    timeout: int = 120
    retry_times: int = 3

7.2 LLM管理器

from typing import Optional, Dict, List
from .base import BaseLLM
from .types import LLMConfig, LLMProvider
from .registry import LLMRegistry

class LLMManager:
    """LLM管理器 - 统一管理LLM实例，支持多实例、热切换、负载均衡"""

    def __init__(self):
        self._instances: Dict[str, BaseLLM] = {}
        self._default: Optional[str] = None

    def add_llm(self, name: str, config: LLMConfig) -> None:
        instance = LLMRegistry.get(config)
        self._instances[name] = instance
        if self._default is None:
            self._default = name

    def set_default(self, name: str) -> None:
        self._default = name

    def get_llm(self, name: Optional[str] = None) -> BaseLLM:
        if name is None:
            name = self._default
        return self._instances[name]

    async def chat(self, messages: List[Dict], **kwargs) -> str:
        return await self.get_llm().chat(messages, **kwargs)

    async def embeddings(self, texts: List[str]) -> List[List[float]]:
        return await self.get_llm().embeddings(texts)
